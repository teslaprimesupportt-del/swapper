   700→// ─── SETTINGS DIALOG ────────────────────────────────
   701→function SettingsDialog() {
   702→  const { providers, setProviders, activeProvider, setActiveProvider, updateProviderStatus } = useStudioStore()
   703→  const [name, setName] = useState('')
   704→  const [type, setType] = useState<ProviderType>('rvc')
   705→  const [endpoint, setEndpoint] = useState('')
   706→  const [apiKey, setApiKey] = useState('')
   707→
   708→  const addProvider = () => {
   709→    if (!name || !endpoint) return
   710→    const newProvider = {
   711→      id: crypto.randomUUID(),
   712→      name, type, endpoint, apiKey,
   713→      status: 'disconnected' as const,
   714→    }
   715→    setProviders([...providers, newProvider])
   716→    setName('')
   717→    setEndpoint('')
   718→    setApiKey('')
   719→  }
   720→
   721→  const testProvider = async (id: string) => {
   722→    const provider = providers.find(p => p.id === id)
   723→    if (!provider) return
   724→    updateProviderStatus(id, 'connecting')
   725→    try {
   726→      const controller = new AbortController()
   727→      const timeout = setTimeout(() => controller.abort(), 5000)
   728→      await fetch(provider.endpoint, {
   729→        method: 'GET',
   730→        signal: controller.signal,
   731→        headers: provider.apiKey ? { Authorization: `Bearer ${provider.apiKey}` } : {},
   732→      })
   733→      clearTimeout(timeout)
   734→      updateProviderStatus(id, 'connected')
   735→    } catch {
   736→      updateProviderStatus(id, 'error')
   737→    }
   738→  }
   739→
   740→  return (
   741→    <Dialog>
   742→      <DialogTrigger asChild>
   743→        <Button variant="ghost" size="icon" className="w-8 h-8">
   744→          <Settings className="w-4 h-4" />
   745→        </Button>
   746→      </DialogTrigger>
   747→      <DialogContent className="sm:max-w-md bg-card border-studio-border/50">
   748→        <DialogHeader>
   749→          <DialogTitle className="flex items-center gap-2">
   750→            <Settings className="w-4 h-4" />
   751→            AI Provider Settings (BYOK)
   752→          </DialogTitle>
   753→        </DialogHeader>
   754→
   755→        <div className="space-y-4 mt-4">
   756→          {/* Add Provider Form */}
   757→          <div className="space-y-3 p-3 rounded-lg border border-studio-border/30 bg-secondary/30">
   758→            <p className="text-xs font-medium uppercase tracking-wider text-studio-muted-foreground/70">
   759→              Add Provider
   760→            </p>
   761→            <div className="grid gap-2">
   762→              <Input
   763→                placeholder="Provider name (e.g. My RVC Server)"
   764→                value={name}
   765→                onChange={(e) => setName(e.target.value)}
   766→                className="h-9 text-sm"
   767→              />
   768→              <Select value={type} onValueChange={(v) => setType(v as ProviderType)}>
   769→                <SelectTrigger className="h-9 text-sm">
   770→                  <SelectValue />
   771→                </SelectTrigger>
   772→                <SelectContent>
   773→                  <SelectItem value="rvc">RVC (Voice Conversion)</SelectItem>
   774→                  <SelectItem value="openvoice">OpenVoice (Voice Clone)</SelectItem>
   775→                  <SelectItem value="liveportrait">LivePortrait (Face)</SelectItem>
   776→                  <SelectItem value="wav2lip">Wav2Lip (Lip Sync)</SelectItem>
   777→                  <SelectItem value="custom">Custom</SelectItem>
   778→                </SelectContent>
   779→              </Select>
   780→              <Input
   781→                placeholder="API endpoint URL"
   782→                value={endpoint}
   783→                onChange={(e) => setEndpoint(e.target.value)}
   784→                className="h-9 text-sm"
   785→              />
   786→              <Input
   787→                placeholder="API Key (optional)"
   788→                type="password"
   789→                value={apiKey}
   790→                onChange={(e) => setApiKey(e.target.value)}
   791→                className="h-9 text-sm"
   792→              />
   793→              <Button onClick={addProvider} size="sm" className="w-full" disabled={!name || !endpoint}>
   794→                Add Provider
   795→              </Button>
   796→            </div>
   797→          </div>
   798→
   799→          {/* Provider List */}
   800→          {providers.length > 0 && (
   801→            <div className="space-y-2">
   802→              <p className="text-xs font-medium uppercase tracking-wider text-studio-muted-foreground/70">
   803→                Configured Providers
   804→              </p>
   805→              {providers.map((p) => (
   806→                <div
   807→                  key={p.id}
   808→                  className={cn(
   809→                    'flex items-center justify-between p-3 rounded-lg border transition-colors',
   810→                    activeProvider?.id === p.id
   811→                      ? 'border-studio-accent/50 bg-studio-accent/5'
   812→                      : 'border-studio-border/30 bg-secondary/30'
   813→                  )}
   814→                >
   815→                  <div className="flex items-center gap-2 min-w-0">
   816→                    <div className={cn(
   817→                      'w-2 h-2 rounded-full shrink-0',
   818→                      p.status === 'connected' ? 'bg-studio-success' :
   819→                      p.status === 'connecting' ? 'bg-studio-warning animate-pulse' :
   820→                      p.status === 'error' ? 'bg-studio-danger' :
   821→                      'bg-studio-muted-foreground/30'
   822→                    )} />
   823→                    <div className="min-w-0">
   824→                      <p className="text-sm font-medium truncate">{p.name}</p>
   825→                      <p className="text-[10px] text-studio-muted-foreground/50 truncate">{p.type} · {p.endpoint}</p>
   826→                    </div>
   827→                  </div>
   828→                  <div className="flex items-center gap-1 shrink-0">
   829→                    <Button
   830→                      variant="ghost"
   831→                      size="sm"
   832→                      className="h-7 text-xs"
   833→                      onClick={() => testProvider(p.id)}
   834→                      disabled={p.status === 'connecting'}
   835→                    >
   836→                      Test
   837→                    </Button>
   838→                    <Button
   839→                      variant="ghost"
   840→                      size="sm"
   841→                      className={cn(
   842→                        'h-7 text-xs',
   843→                        activeProvider?.id === p.id && 'text-studio-accent'
   844→                      )}
   845→                      onClick={() => setActiveProvider(p)}
   846→                    >
   847→                      {activeProvider?.id === p.id ? 'Active' : 'Use'}
   848→                    </Button>
   849→                  </div>
   850→                </div>
   851→              ))}
   852→            </div>
   853→          )}
   854→
   855→          {providers.length === 0 && (
   856→            <p className="text-center text-xs text-studio-muted-foreground/40 py-4">
   857→              No providers configured yet. Add your RVC, OpenVoice, or custom AI endpoint above.
   858→            </p>
   859→          )}
   860→        </div>
   861→      </DialogContent>
   862→    </Dialog>
   863→  )
   864→}
