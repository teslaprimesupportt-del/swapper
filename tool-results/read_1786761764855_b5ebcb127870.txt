   700→  const [apiKey, setApiKey] = useState('')
   701→
   702→  const addProvider = () => {
   703→    if (!name || !endpoint) return
   704→    const newProvider = {
   705→      id: crypto.randomUUID(),
   706→      name, type, endpoint, apiKey,
   707→      status: 'disconnected' as const,
   708→    }
   709→    setProviders([...providers, newProvider])
   710→    setName('')
   711→    setEndpoint('')
   712→    setApiKey('')
   713→  }
   714→
   715→  const testProvider = async (id: string) => {
   716→    const provider = providers.find(p => p.id === id)
   717→    if (!provider) return
   718→    updateProviderStatus(id, 'connecting')
   719→    try {
   720→      const controller = new AbortController()
   721→      const timeout = setTimeout(() => controller.abort(), 5000)
   722→      await fetch(provider.endpoint, {
   723→        method: 'GET',
   724→        signal: controller.signal,
   725→        headers: provider.apiKey ? { Authorization: `Bearer ${provider.apiKey}` } : {},
   726→      })
   727→      clearTimeout(timeout)
   728→      updateProviderStatus(id, 'connected')
   729→    } catch {
   730→      updateProviderStatus(id, 'error')
   731→    }
   732→  }
   733→
   734→  return (
   735→    <Dialog>
   736→      <DialogTrigger asChild>
   737→        <Button variant="ghost" size="icon" className="w-8 h-8">
   738→          <Settings className="w-4 h-4" />
   739→        </Button>
   740→      </DialogTrigger>
   741→      <DialogContent className="sm:max-w-md bg-card border-studio-border/50">
   742→        <DialogHeader>
   743→          <DialogTitle className="flex items-center gap-2">
   744→            <Settings className="w-4 h-4" />
   745→            AI Provider Settings (BYOK)
   746→          </DialogTitle>
   747→        </DialogHeader>
   748→
   749→        <div className="space-y-4 mt-4">
   750→          {/* Add Provider Form */}
   751→          <div className="space-y-3 p-3 rounded-lg border border-studio-border/30 bg-secondary/30">
   752→            <p className="text-xs font-medium uppercase tracking-wider text-studio-muted-foreground/70">
   753→              Add Provider
   754→            </p>
   755→            <div className="grid gap-2">
   756→              <Input
   757→                placeholder="Provider name (e.g. My RVC Server)"
   758→                value={name}
   759→                onChange={(e) => setName(e.target.value)}
   760→                className="h-9 text-sm"
   761→              />
   762→              <Select value={type} onValueChange={(v) => setType(v as ProviderType)}>
   763→                <SelectTrigger className="h-9 text-sm">
   764→                  <SelectValue />
   765→                </SelectTrigger>
   766→                <SelectContent>
   767→                  <SelectItem value="rvc">RVC (Voice Conversion)</SelectItem>
   768→                  <SelectItem value="openvoice">OpenVoice (Voice Clone)</SelectItem>
   769→                  <SelectItem value="liveportrait">LivePortrait (Face)</SelectItem>
   770→                  <SelectItem value="wav2lip">Wav2Lip (Lip Sync)</SelectItem>
   771→                  <SelectItem value="custom">Custom</SelectItem>
   772→                </SelectContent>
   773→              </Select>
   774→              <Input
   775→                placeholder="API endpoint URL"
   776→                value={endpoint}
   777→                onChange={(e) => setEndpoint(e.target.value)}
   778→                className="h-9 text-sm"
   779→              />
   780→              <Input
   781→                placeholder="API Key (optional)"
   782→                type="password"
   783→                value={apiKey}
   784→                onChange={(e) => setApiKey(e.target.value)}
   785→                className="h-9 text-sm"
   786→              />
   787→              <Button onClick={addProvider} size="sm" className="w-full" disabled={!name || !endpoint}>
   788→                Add Provider
   789→              </Button>
   790→            </div>
   791→          </div>
   792→
   793→          {/* Provider List */}
   794→          {providers.length > 0 && (
   795→            <div className="space-y-2">
   796→              <p className="text-xs font-medium uppercase tracking-wider text-studio-muted-foreground/70">
   797→                Configured Providers
   798→              </p>
   799→              {providers.map((p) => (
   800→                <div
   801→                  key={p.id}
   802→                  className={cn(
   803→                    'flex items-center justify-between p-3 rounded-lg border transition-colors',
   804→                    activeProvider?.id === p.id
   805→                      ? 'border-studio-accent/50 bg-studio-accent/5'
   806→                      : 'border-studio-border/30 bg-secondary/30'
   807→                  )}
   808→                >
   809→                  <div className="flex items-center gap-2 min-w-0">
   810→                    <div className={cn(
   811→                      'w-2 h-2 rounded-full shrink-0',
   812→                      p.status === 'connected' ? 'bg-studio-success' :
   813→                      p.status === 'connecting' ? 'bg-studio-warning animate-pulse' :
   814→                      p.status === 'error' ? 'bg-studio-danger' :
   815→                      'bg-studio-muted-foreground/30'
   816→                    )} />
   817→                    <div className="min-w-0">
   818→                      <p className="text-sm font-medium truncate">{p.name}</p>
   819→                      <p className="text-[10px] text-studio-muted-foreground/50 truncate">{p.type} · {p.endpoint}</p>
   820→                    </div>
   821→                  </div>
   822→                  <div className="flex items-center gap-1 shrink-0">
   823→                    <Button
   824→                      variant="ghost"
   825→                      size="sm"
   826→                      className="h-7 text-xs"
   827→                      onClick={() => testProvider(p.id)}
   828→                      disabled={p.status === 'connecting'}
   829→                    >
   830→                      Test
   831→                    </Button>
   832→                    <Button
   833→                      variant="ghost"
   834→                      size="sm"
   835→                      className={cn(
   836→                        'h-7 text-xs',
   837→                        activeProvider?.id === p.id && 'text-studio-accent'
   838→                      )}
   839→                      onClick={() => setActiveProvider(p)}
   840→                    >
   841→                      {activeProvider?.id === p.id ? 'Active' : 'Use'}
   842→                    </Button>
   843→                  </div>
   844→                </div>
   845→              ))}
   846→            </div>
   847→          )}
   848→
   849→          {providers.length === 0 && (
   850→            <p className="text-center text-xs text-studio-muted-foreground/40 py-4">
   851→              No providers configured yet. Add your RVC, OpenVoice, or custom AI endpoint above.
   852→            </p>
   853→          )}
   854→        </div>
   855→      </DialogContent>
   856→    </Dialog>
   857→  )
   858→}
