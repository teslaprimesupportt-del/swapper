    30→    30→function formatDuration(seconds: number): string {
    31→    31→  const h = Math.floor(seconds / 3600)
    32→    32→  const m = Math.floor((seconds % 3600) / 60)
    33→    33→  const s = seconds % 60
    34→    34→  if (h > 0) return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    35→    35→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    36→    36→}
