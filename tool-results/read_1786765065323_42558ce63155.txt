   625→   668→  const { providers, setProviders, activeProvider, setActiveProvider, updateProviderStatus } = useStudioStore()
   626→   669→  const [name, setName] = useState('')
   627→   670→  const [type, setType] = useState<ProviderType>('rvc')
   628→   671→  const [endpoint, setEndpoint] = useState('')
   629→   672→  const [apiKey, setApiKey] = useState('')
   630→   673→
   631→   674→  const addProvider = () => {
   632→   675→    if (!name || !endpoint) return
   633→   676→    const newProvider = {
   634→   677→      id: crypto.randomUUID(),
   635→   678→      name, type, endpoint, apiKey,
   636→   679→      status: 'disconnected' as const,
   637→   680→    }
   638→   681→    setProviders([...providers, newProvider])

   640→   683→    setEndpoint('')
   641→   684→    setApiKey('')
   642→   685→  }
   643→   686→
   644→   687→  const testProvider = async (id: string) => {
   645→   688→    const provider = providers.find(p => p.id === id)
   646→   689→    if (!provider) return
   647→   690→    updateProviderStatus(id, 'connecting')
   648→   691→    try {
   649→   692→      const controller = new AbortController()
   650→   693→      const timeout = setTimeout(() => controller.abort(), 5000)
   651→   694→      await fetch(provider.endpoint, {
   652→   695→        method: 'GET',
   653→   696→        signal: controller.signal,
   654→   697→        headers: provider.apiKey ? { Authorization: `Bearer ${provider.apiKey}` } : {},
   655→   698→      })
   656→   699→      clearTimeout(timeout)
   657→   700→      updateProviderStatus(id, 'connected')
   658→   701→    } catch {
   659→   702→      updateProviderStatus(id, 'error')
