   678→      name, type, endpoint, apiKey,
   679→      status: 'disconnected' as const,
   680→    }
   681→    setProviders([...providers, newProvider])
   682→    setName('')
   683→    setEndpoint('')
   684→    setApiKey('')
   685→  }
   686→
   687→  const testProvider = async (id: string) => {
   688→    const provider = providers.find(p => p.id === id)
   689→    if (!provider) return
   690→    updateProviderStatus(id, 'connecting')
   691→    try {
   692→      const controller = new AbortController()
   693→      const timeout = setTimeout(() => controller.abort(), 5000)
   694→      await fetch(provider.endpoint, {
   695→        method: 'GET',
   696→        signal: controller.signal,
   697→        headers: provider.apiKey ? { Authorization: `Bearer ${provider.apiKey}` } : {},
   698→      })
   699→      clearTimeout(timeout)
   700→      updateProviderStatus(id, 'connected')
   701→    } catch {
   702→      updateProviderStatus(id, 'error')
   703→    }
   704→  }
   705→
   706→  return (
   707→    <Dialog>
   708→      <DialogTrigger asChild>
   709→        <Button variant="ghost" size="icon" className="w-8 h-8">
   710→          <Settings className="w-4 h-4" />
   711→        </Button>
   712→      </DialogTrigger>
   713→      <DialogContent className="sm:max-w-md bg-card border-studio-border/50">

   715→          <DialogTitle className="flex items-center gap-2">
   716→            <Settings className="w-4 h-4" />
   717→            AI Provider Settings (BYOK)
   718→          </DialogTitle>
   719→        </DialogHeader>
   720→
   721→        <div className="space-y-4 mt-4">
   722→          {/* Add Provider Form */}
   723→          <div className="space-y-3 p-3 rounded-lg border border-studio-border/30 bg-secondary/30">
   724→            <p className="text-xs font-medium uppercase tracking-wider text-studio-muted-foreground/70">
   725→              Add Provider
   726→            </p>
   727→            <div className="grid gap-2">
   728→              <Input
   729→                placeholder="Provider name (e.g. My RVC Server)"
   730→                value={name}
   731→                onChange={(e) => setName(e.target.value)}
   732→                className="h-9 text-sm"
   733→              />
   734→              <Select value={type} onValueChange={(v) => setType(v as ProviderType)}>
   735→                <SelectTrigger className="h-9 text-sm">
   736→                  <SelectValue />
   737→                </SelectTrigger>
   738→                <SelectContent>
   739→                  <SelectItem value="rvc">RVC (Voice Conversion)</SelectItem>
   740→                  <SelectItem value="openvoice">OpenVoice (Voice Clone)</SelectItem>
   741→                  <SelectItem value="liveportrait">LivePortrait (Face)</SelectItem>
   742→                  <SelectItem value="wav2lip">Wav2Lip (Lip Sync)</SelectItem>
   743→                  <SelectItem value="custom">Custom</SelectItem>
   744→                </SelectContent>
   745→              </Select>
   746→              <Input
   747→                placeholder="API endpoint URL"
   748→                value={endpoint}
   749→                onChange={(e) => setEndpoint(e.target.value)}
   750→                className="h-9 text-sm"
   751→              />
   752→              <Input
   753→                placeholder="API Key (optional)"
   754→                type="password"
   755→                value={apiKey}
   756→                onChange={(e) => setApiKey(e.target.value)}
   757→                className="h-9 text-sm"
   758→              />
   759→              <Button onClick={addProvider} size="sm" className="w-full" disabled={!name || !endpoint}>
   760→                Add Provider
   761→              </Button>
   762→            </div>
   763→          </div>
   764→
   765→          {/* Provider List */}
   766→          {providers.length > 0 && (
   767→            <div className="space-y-2">
   768→              <p className="text-xs font-medium uppercase tracking-wider text-studio-muted-foreground/70">
   769→                Configured Providers
   770→              </p>
   771→              {providers.map((p) => (
   772→                <div
   773→                  key={p.id}
   774→                  className={cn(
   775→                    'flex items-center justify-between p-3 rounded-lg border transition-colors',
   776→                    activeProvider?.id === p.id
   777→                      ? 'border-studio-accent/50 bg-studio-accent/5'
   778→                      : 'border-studio-border/30 bg-secondary/30'
   779→                  )}
   780→                >
   781→                  <div className="flex items-center gap-2 min-w-0">
   782→                    <div className={cn(
   783→                      'w-2 h-2 rounded-full shrink-0',
   784→                      p.status === 'connected' ? 'bg-studio-success' :
   785→                      p.status === 'connecting' ? 'bg-studio-warning animate-pulse' :
   786→                      p.status === 'error' ? 'bg-studio-danger' :
   787→                      'bg-studio-muted-foreground/30'
   788→                    )} />
   789→                    <div className="min-w-0">
   790→                      <p className="text-sm font-medium truncate">{p.name}</p>
   791→                      <p className="text-[10px] text-studio-muted-foreground/50 truncate">{p.type} · {p.endpoint}</p>
   792→                    </div>
   793→                  </div>
   794→                  <div className="flex items-center gap-1 shrink-0">

   799→                      onClick={() => testProvider(p.id)}

   801→                    >
   802→                      Test
   803→                    </Button>
   804→                    <Button
   805→                      variant="ghost"
   806→                      size="sm"
   807→                      className={cn(
   808→                        'h-7 text-xs',
   809→                        activeProvider?.id === p.id && 'text-studio-accent'
   810→                      )}
   811→                      onClick={() => setActiveProvider(p)}
   812→                    >
   813→                      {activeProvider?.id === p.id ? 'Active' : 'Use'}
   814→                    </Button>
   815→                  </div>
   816→                </div>
   817→              ))}
   818→            </div>
   819→          )}
   820→
   821→          {providers.length === 0 && (
   822→            <p className="text-center text-xs text-studio-muted-foreground/40 py-4">
   823→              No providers configured yet. Add your RVC, OpenVoice, or custom AI endpoint above.
   824→            </p>
   825→          )}
   826→        </div>
   827→      </DialogContent>
   828→    </Dialog>
   829→  )
