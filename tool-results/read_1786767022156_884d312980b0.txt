    32→    32→function formatDuration(seconds: number): string {
    33→    33→  const h = Math.floor(seconds / 3600)
    34→    34→  const m = Math.floor((seconds % 3600) / 60)
    35→    35→  const s = seconds % 60
    36→    36→  if (h > 0) return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    37→    37→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    38→    38→}
