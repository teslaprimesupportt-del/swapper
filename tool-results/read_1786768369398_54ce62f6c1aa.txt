    35→    35→    35→function formatDuration(seconds: number): string {
    36→    36→    36→  const h = Math.floor(seconds / 3600)
    37→    37→    37→  const m = Math.floor((seconds % 3600) / 60)
    38→    38→    38→  const s = seconds % 60
    39→    39→    39→  if (h > 0) return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    40→    40→    40→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    41→    41→    41→}
