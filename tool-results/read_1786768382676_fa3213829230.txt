     1→     1→   501→   501→   501→    const resampled = resampleAudio(mono, audioBuffer.sampleRate, targetRate)
     2→     2→   502→   502→   502→
     3→     3→   503→   503→   503→    // 10 second chunks at model sample rate
     4→     4→   504→   504→   504→    const chunkDuration = 10 * targetRate
     5→     5→   505→   505→   505→    const chunks: Float32Array[] = []
     6→     6→   506→   506→   506→
     7→     7→   507→   507→   507→    for (let start = 0; start < resampled.length; start += chunkDuration) {
     8→     8→   508→   508→   508→      const end = Math.min(start + chunkDuration, resampled.length)
     9→     9→   509→   509→   509→      const chunk = resampled.slice(start, end)
    10→    10→   510→   510→   510→      chunks.push(chunk)
    11→    11→   511→   511→   511→    }
    12→    12→   512→   512→   512→
    13→    13→   513→   513→   513→    // Process chunks sequentially (could parallelize for offline)
    14→    14→   514→   514→   514→    const convertedChunks: Float32Array[] = []
    15→    15→   515→   515→   515→    for (const chunk of chunks) {
    16→    16→   516→   516→   516→      const converted = await this.convertChunk(chunk, targetRate)
    17→    17→   517→   517→   517→      convertedChunks.push(converted)
    18→    18→   518→   518→   518→    }
    19→    19→   519→   519→   519→
    20→    20→   520→   520→   520→    // Concatenate all converted chunks
    21→    21→   521→   521→   521→    const totalLength = convertedChunks.reduce((sum, c) => sum + c.length, 0)
    22→    22→   522→   522→   522→    const result = new Float32Array(totalLength)
    23→    23→   523→   523→   523→    let offset = 0
    24→    24→   524→   524→   524→    for (const chunk of convertedChunks) {
    25→    25→   525→   525→   525→      result.set(chunk, offset)
    26→    26→   526→   526→   526→      offset += chunk.length
    27→    27→   527→   527→   527→    }
    28→    28→   528→   528→   528→
    29→    29→   529→   529→   529→    return result
    30→    30→   530→   530→   530→  }

   123→   123→   623→   623→   623→   * Seed-VC's Gradio app typically exposes endpoints like:
   124→   124→   624→   624→   624→   *   - /api/predict (legacy sync)
   125→   125→   625→   625→   625→   *   - /call/{fn_name} (async)
   126→   126→   626→   626→   626→   * 
   127→   127→   627→   627→   627→   * We detect the available endpoint from the schema and use it.
   128→   128→   628→   628→   628→   * The function sends source WAV + reference WAV and receives converted WAV.
   129→   129→   629→   629→   629→   */

   131→   131→   631→   631→   631→    sourceBase64: string,
   132→   132→   632→   632→   632→    referenceBase64: string,
   133→   133→   633→   633→   633→    sourceName: string,

   137→   137→   637→   637→   637→    const schema = this.state.apiSchema
   138→   138→   638→   638→   638→
   139→   139→   639→   639→   639→    // Strategy 1: Try named endpoints (Gradio 4.x /call/{name} pattern)
   140→   140→   640→   640→   640→    if (schema?.named_endpoints) {
   141→   141→   641→   641→   641→      const endpointNames = Object.keys(schema.named_endpoints)
   142→   142→   642→   642→   642→      // Look for VC-related endpoints
   143→   143→   643→   643→   643→      const vcEndpoint = endpointNames.find(n =>
   144→   144→   644→   644→   644→        /vc|voice|convert|real.?time/i.test(n)
   145→   145→   645→   645→   645→      ) || endpointNames[0]
   146→   146→   646→   646→   646→
   147→   147→   647→   647→   647→      if (vcEndpoint) {
   148→   148→   648→   648→   648→        return await this.callNamedEndpoint(baseUrl, vcEndpoint, sourceBase64, referenceBase64)
   149→   149→   649→   649→   649→      }
   150→   150→   650→   650→   650→    }
   151→   151→   651→   651→   651→
   152→   152→   652→   652→   652→    // Strategy 2: Fall back to legacy /api/predict
   153→   153→   653→   653→   653→    return await this.callLegacyPredict(baseUrl, sourceBase64, referenceBase64)
   154→   154→   654→   654→   654→  }

   156→   156→   656→   656→   656→  /**
   157→   157→   657→   657→   657→   * Call a named Gradio endpoint using the async /call/{name} pattern.
   158→   158→   658→   658→   658→   * POST /call/{name} -> get event_id -> GET /call/{name}/{event_id} -> poll for result
   159→   159→   659→   659→   659→   */
   160→   160→   660→   660→   660→  private async callNamedEndpoint(
   161→   161→   661→   661→   661→    baseUrl: string,
   162→   162→   662→   662→   662→    endpointName: string,
   163→   163→   663→   663→   663→    sourceBase64: string,
   164→   164→   664→   664→   664→    referenceBase64: string
   165→   165→   665→   665→   665→  ): Promise<ArrayBuffer> {
   166→   166→   666→   666→   666→    const url = `${baseUrl}/call/${endpointName}`
   167→   167→   667→   667→   667→
   168→   168→   668→   668→   668→    // Step 1: Submit the job
   169→   169→   669→   669→   669→    const submitResponse = await fetch(url, {
   170→   170→   670→   670→   670→      method: 'POST',
   171→   171→   671→   671→   671→      headers: {
   172→   172→   672→   672→   672→        ...this.getHeaders(),
   173→   173→   673→   673→   673→        'Content-Type': 'application/json',
   174→   174→   674→   674→   674→      },
   175→   175→   675→   675→   675→      body: JSON.stringify({
   176→   176→   676→   676→   676→        // Gradio expects data as an array of parameter values
   177→   177→   677→   677→   677→        // The order must match the endpoint's parameter list
   178→   178→   678→   678→   678→        data: [
   179→   179→   679→   679→   679→          { data: sourceBase64, meta: { _type: 'gradio.FileData' } },
   180→   180→   680→   680→   680→          { data: referenceBase64, meta: { _type: 'gradio.FileData' } },
   181→   181→   681→   681→   681→          // Additional params depend on the Gradio app's interface
   182→   182→   682→   682→   682→          this.config.diffusionSteps,
   183→   183→   683→   683→   683→          this.config.cfgRate,
   184→   184→   684→   684→   684→          this.config.semiToneShift,
   185→   185→   685→   685→   685→          this.config.lengthAdjust,
   186→   186→   686→   686→   686→          this.config.f0Condition,
   187→   187→   687→   687→   687→          this.config.autoF0Adjust,
   188→   188→   688→   688→   688→        ],
   189→   189→   689→   689→   689→      }),
   190→   190→   690→   690→   690→      signal: this.getSignal(),
   191→   191→   691→   691→   691→    })
   192→   192→   692→   692→   692→
   193→   193→   693→   693→   693→    if (!submitResponse.ok) {
   194→   194→   694→   694→   694→      throw new Error(`Gradio submit failed: ${submitResponse.status} ${submitResponse.statusText}`)
   195→   195→   695→   695→   695→    }
   196→   196→   696→   696→   696→
   197→   197→   697→   697→   697→    const submitData = await submitResponse.json()
   198→   198→   698→   698→   698→    const eventId = submitData.event_id
   199→   199→   699→   699→   699→
   200→   200→   700→   700→   700→    if (!eventId) {
   201→   201→   701→   701→   701→      // If no event_id, try treating response as direct result (sync mode)
   202→   202→   702→   702→   702→      if (submitData.data) {

   204→   204→   704→   704→   704→      }
   205→   205→   705→   705→   705→      throw new Error('Gradio did not return an event_id')
   206→   206→   706→   706→   706→    }
   207→   207→   707→   707→   707→
   208→   208→   708→   708→   708→    // Step 2: Poll for the result

   210→   210→   710→   710→   710→  }
   211→   211→   711→   711→   711→
   212→   212→   712→   712→   712→  /**
   213→   213→   713→   713→   713→   * Poll the Gradio event endpoint until the job completes.
   214→   214→   714→   714→   714→   * Gradio sends SSE-style events: 'processing', 'complete', 'error'.
   215→   215→   715→   715→   715→   */
   216→   216→   716→   716→   716→  private async pollForResult(
   217→   217→   717→   717→   717→    baseUrl: string,
   218→   218→   718→   718→   718→    endpointName: string,
   219→   219→   719→   719→   719→    eventId: string
   220→   220→   720→   720→   720→  ): Promise<ArrayBuffer> {
   221→   221→   721→   721→   721→    const url = `${baseUrl}/call/${endpointName}/${eventId}`
   222→   222→   722→   722→   722→    const maxPollTime = 30_000 // 30 second timeout for polling
   223→   223→   723→   723→   723→    const pollInterval = 200   // poll every 200ms
   224→   224→   724→   724→   724→    const startTime = Date.now()
   225→   225→   725→   725→   725→
   226→   226→   726→   726→   726→    while (Date.now() - startTime < maxPollTime) {
   227→   227→   727→   727→   727→      const response = await fetch(url, {
   228→   228→   728→   728→   728→        method: 'GET',
   229→   229→   729→   729→   729→        headers: this.getHeaders(),
   230→   230→   730→   730→   730→        signal: this.getSignal(),
   231→   231→   731→   731→   731→      })
   232→   232→   732→   732→   732→
   233→   233→   733→   733→   733→      if (!response.ok) {
   234→   234→   734→   734→   734→        throw new Error(`Gradio poll failed: ${response.status}`)
   235→   235→   735→   735→   735→      }
   236→   236→   736→   736→   736→
   237→   237→   737→   737→   737→      const event = await response.json()
   238→   238→   738→   738→   738→
   239→   239→   739→   739→   739→      switch (event.msg) {
   240→   240→   740→   740→   740→        case 'processing':
   241→   241→   741→   741→   741→          // Still working, continue polling
   242→   242→   742→   742→   742→          break
   243→   243→   743→   743→   743→
   244→   244→   744→   744→   744→        case 'complete':
   245→   245→   745→   745→   745→          return await this.extractAudioFromResponse(event.output, baseUrl)
   246→   246→   746→   746→   746→
   247→   247→   747→   747→   747→        case 'error':
   248→   248→   748→   748→   748→          throw new Error(`Seed-VC inference error: ${event.output?.error || 'Unknown error'}`)
   249→   249→   749→   749→   749→
   250→   250→   750→   750→   750→        default:
   251→   251→   751→   751→   751→          // Unknown message type, keep polling
   252→   252→   752→   752→   752→          break
   253→   253→   753→   753→   753→      }
   254→   254→   754→   754→   754→
   255→   255→   755→   755→   755→      await new Promise(resolve => setTimeout(resolve, pollInterval))
   256→   256→   756→   756→   756→    }
   257→   257→   757→   757→   757→
   258→   258→   758→   758→   758→    throw new Error(`Seed-VC inference timed out after ${maxPollTime / 1000}s`)
   259→   259→   759→   759→   759→  }

   299→   299→   799→   799→   799→   * Gradio returns file info with a URL; we download the actual WAV data.
   300→   300→   800→   800→   800→   */
   301→   301→   801→   801→   801→  private async extractAudioFromResponse(
   302→   302→   802→   802→   802→    response: { data?: unknown[]; output?: unknown[] },
   303→   303→   803→   803→   803→    baseUrl: string
   304→   304→   804→   804→   804→  ): Promise<ArrayBuffer> {
   305→   305→   805→   805→   805→    const dataArray = response.data || response.output
   306→   306→   806→   806→   806→    if (!dataArray || dataArray.length === 0) {
   307→   307→   807→   807→   807→      throw new Error('Gradio returned empty data')
   308→   308→   808→   808→   808→    }
   309→   309→   809→   809→   809→
   310→   310→   810→   810→   810→    const firstResult = dataArray[0]
   311→   311→   811→   811→   811→
   312→   312→   812→   812→   812→    // Case 1: Direct base64 data (some Gradio configs return inline data)
   313→   313→   813→   813→   813→    if (typeof firstResult === 'string' && firstResult.startsWith('data:')) {
   314→   314→   814→   814→   814→      const base64Data = firstResult.split(',')[1]
   315→   315→   815→   815→   815→      if (base64Data) {
   316→   316→   816→   816→   816→        return base64ToArrayBuffer(base64Data)
   317→   317→   817→   817→   817→      }
   318→   318→   818→   818→   818→    }
   319→   319→   819→   819→   819→
   320→   320→   820→   820→   820→    // Case 2: File URL (standard Gradio behavior)
   321→   321→   821→   821→   821→    if (firstResult && typeof firstResult === 'object') {
   322→   322→   822→   822→   822→      const fileInfo = firstResult as GradioFileInfo
   323→   323→   823→   823→   823→      let fileUrl = fileInfo.url || ''
   324→   324→   824→   824→   824→
   325→   325→   825→   825→   825→      // Handle relative URLs
   326→   326→   826→   826→   826→      if (fileUrl.startsWith('/')) {
   327→   327→   827→   827→   827→        fileUrl = `${baseUrl}${fileUrl}`
   328→   328→   828→   828→   828→      }
   329→   329→   829→   829→   829→
   330→   330→   830→   830→   830→      if (!fileUrl) {
   331→   331→   831→   831→   831→        throw new Error('Gradio file response has no URL')
   332→   332→   832→   832→   832→      }
   333→   333→   833→   833→   833→
   334→   334→   834→   834→   834→      const audioResponse = await fetch(fileUrl, {
   335→   335→   835→   835→   835→        headers: this.getHeaders(),
   336→   336→   836→   836→   836→        signal: this.getSignal(),
   337→   337→   837→   837→   837→      })
   338→   338→   838→   838→   838→
   339→   339→   839→   839→   839→      if (!audioResponse.ok) {
   340→   340→   840→   840→   840→        throw new Error(`Failed to download converted audio: ${audioResponse.status}`)
   341→   341→   841→   841→   841→      }
   342→   342→   842→   842→   842→
   343→   343→   843→   843→   843→      return await audioResponse.arrayBuffer()
   344→   344→   844→   844→   844→    }
   345→   345→   845→   845→   845→
   346→   346→   846→   846→   846→    throw new Error(`Unexpected Gradio response format: ${typeof firstResult}`)
   347→   347→   847→   847→   847→  }
   348→   348→   848→   848→   848→

   352→   352→   852→   852→   852→    }

   355→   355→   855→   855→   855→    }

   373→   373→   873→   873→   873→}

   407→   407→   907→   907→   907→/**
   408→   408→   908→   908→   908→ * Seed-VC Real-Time Pipeline
   409→   409→   909→   909→   909→ *

   411→   411→   911→   911→   911→ *   1. Captures chunks from the microphone via ScriptProcessorNode/AudioWorklet

   417→   417→   917→   917→   917→ *   const pipeline = new SeedVCRealtimePipeline(adapter, audioContext, config)

   421→   421→   921→   921→   921→ */

   423→   423→   923→   923→   923→  private adapter: SeedVCAdapter
   424→   424→   924→   924→   924→  private audioContext: AudioContext

   426→   426→   926→   926→   926→  private isRunning: boolean = false
   427→   427→   927→   927→   927→
   428→   428→   928→   928→   928→  // Audio nodes
   429→   429→   929→   929→   929→  private sourceNode: MediaStreamAudioSourceNode | null = null
   430→   430→   930→   930→   930→  private processorNode: ScriptProcessorNode | null = null

   432→   432→   932→   932→   932→
   433→   433→   933→   933→   933→  // Buffers
   434→   434→   934→   934→   934→  private inputRingBuffer: Float32Array = new Float32Array(0)
   435→   435→   935→   935→   935→  private readonly maxBufferSize: number
   436→   436→   936→   936→   936→  private playbackQueue: Float32Array[] = []
   437→   437→   937→   937→   937→  private isPlaying: boolean = false
   438→   438→   938→   938→   938→
   439→   439→   939→   939→   939→  // Callbacks

   442→   442→   942→   942→   942→
   443→   443→   943→   943→   943→  constructor(

   451→   451→   951→   951→   951→    // Max buffer = left context + right context + chunk, with 2x safety margin

   453→   453→   953→   953→   953→      (this.config.leftContext + this.config.rightContext + this.config.chunkDuration) * 2
   454→   454→   954→   954→   954→      * this.config.sourceSampleRate
   455→   455→   955→   955→   955→    )
   456→   456→   956→   956→   956→  }
   457→   457→   957→   957→   957→
   458→   458→   958→   958→   958→  /**
   459→   459→   959→   959→   959→   * Start the real-time conversion pipeline.
   460→   460→   960→   960→   960→   * Connects to the microphone and begins processing audio chunks.
   461→   461→   961→   961→   961→   */
   462→   462→   962→   962→   962→  async start(stream: MediaStream): Promise<void> {
   463→   463→   963→   963→   963→    if (this.isRunning) return
   464→   464→   964→   964→   964→    this.isRunning = true
   465→   465→   965→   965→   965→
   466→   466→   966→   966→   966→    // Create audio nodes
   467→   467→   967→   967→   967→    this.sourceNode = this.audioContext.createMediaStreamSource(stream)
   468→   468→   968→   968→   968→    this.gainNode = this.audioContext.createGain()
   469→   469→   969→   969→   969→    this.gainNode.gain.value = 1.0
   470→   470→   970→   970→   970→
   471→   471→   971→   971→   971→    // ScriptProcessorNode for capturing raw PCM (deprecated but universally supported)
   472→   472→   972→   972→   972→    // Buffer size = chunk duration in samples
   473→   473→   973→   973→   973→    const bufferSize = Math.ceil(
   474→   474→   974→   974→   974→      this.config.chunkDuration * this.config.sourceSampleRate / 128
   475→   475→   975→   975→   975→    ) * 128 // Round up to power-of-2 multiple
   476→   476→   976→   976→   976→    this.processorNode = this.audioContext.createScriptProcessor(bufferSize, 1, 1)
   477→   477→   977→   977→   977→
   478→   478→   978→   978→   978→    this.processorNode.onaudioprocess = (event) => {
   479→   479→   979→   979→   979→      if (!this.isRunning) return
   480→   480→   980→   980→   980→      const inputData = event.inputBuffer.getChannelData(0)
   481→   481→   981→   981→   981→      this.processInputChunk(inputData)
   482→   482→   982→   982→   982→    }
   483→   483→   983→   983→   983→
   484→   484→   984→   984→   984→    // Connect: mic -> processor (for capture) -> gain -> destination (for monitoring)
   485→   485→   985→   985→   985→    // We don't connect processor to gain to avoid feedback loop
   486→   486→   986→   986→   986→    this.sourceNode.connect(this.processorNode)
   487→   487→   987→   987→   987→    // processor is NOT connected to destination to prevent echo
   488→   488→   988→   988→   988→  }
   489→   489→   989→   989→   989→
   490→   490→   990→   990→   990→  /**
   491→   491→   991→   991→   991→   * Stop the real-time pipeline and release resources.
   492→   492→   992→   992→   992→   */
   493→   493→   993→   993→   993→  stop(): void {
   494→   494→   994→   994→   994→    this.isRunning = false
   495→   495→   995→   995→   995→
   496→   496→   996→   996→   996→    if (this.processorNode) {
   497→   497→   997→   997→   997→      this.processorNode.onaudioprocess = null
   498→   498→   998→   998→   998→      this.processorNode.disconnect()
   499→   499→   999→   999→   999→      this.processorNode = null
   500→   500→  1000→  1000→  1000→    }
   501→   501→  1001→  1001→  1001→    if (this.sourceNode) {
   502→   502→  1002→  1002→  1002→      this.sourceNode.disconnect()
   503→   503→  1003→  1003→  1003→      this.sourceNode = null
   504→   504→  1004→  1004→  1004→    }
   505→   505→  1005→  1005→  1005→    if (this.gainNode) {
   506→   506→  1006→  1006→  1006→      this.gainNode.disconnect()
   507→   507→  1007→  1007→  1007→      this.gainNode = null
   508→   508→  1008→  1008→  1008→    }
   509→   509→  1009→  1009→  1009→
   510→   510→  1010→  1010→  1010→    this.inputRingBuffer = new Float32Array(0)
   511→   511→  1011→  1011→  1011→    this.playbackQueue = []
   512→   512→  1012→  1012→  1012→    this.isPlaying = false
   513→   513→  1013→  1013→  1013→  }
   514→   514→  1014→  1014→  1014→
   515→   515→  1015→  1015→  1015→  /**
   516→   516→  1016→  1016→  1016→   * Register a callback for converted audio output.
   517→   517→  1017→  1017→  1017→   * The callback receives Float32Array samples ready for playback.
   518→   518→  1018→  1018→  1018→   */
   519→   519→  1019→  1019→  1019→  setOnAudioOutput(callback: (samples: Float32Array) => void): void {
   520→   520→  1020→  1020→  1020→    this.onAudioOutput = callback
   521→   521→  1021→  1021→  1021→  }
   522→   522→  1022→  1022→  1022→
   523→   523→  1023→  1023→  1023→  /**
   524→   524→  1024→  1024→  1024→   * Register a callback for errors.
   525→   525→  1025→  1025→  1025→   */
   526→   526→  1026→  1026→  1026→  setOnError(callback: (error: Error) => void): void {
   527→   527→  1027→  1027→  1027→    this.onError = callback
   528→   528→  1028→  1028→  1028→  }
   529→   529→  1029→  1029→  1029→
   530→   530→  1030→  1030→  1030→  /**
   531→   531→  1031→  1031→  1031→   * Play converted audio through the speakers.
   532→   532→  1032→  1032→  1032→   * Creates a temporary AudioBufferSourceNode for each chunk.
   533→   533→  1033→  1033→  1033→   */
   534→   534→  1034→  1034→  1034→  playConvertedAudio(samples: Float32Array): void {
   535→   535→  1035→  1035→  1035→    const sampleRate = this.config.sourceSampleRate
   536→   536→  1036→  1036→  1036→    const audioBuffer = this.audioContext.createBuffer(1, samples.length, sampleRate)
   537→   537→  1037→  1037→  1037→    audioBuffer.copyToChannel(samples as Float32Array<ArrayBuffer>, 0)
   538→   538→  1038→  1038→  1038→
   539→   539→  1039→  1039→  1039→    const source = this.audioContext.createBufferSource()
   540→   540→  1040→  1040→  1040→    source.buffer = audioBuffer
   541→   541→  1041→  1041→  1041→
   542→   542→  1042→  1042→  1042→    const gainNode = this.audioContext.createGain()
   543→   543→  1043→  1043→  1043→    gainNode.gain.value = 1.0
   544→   544→  1044→  1044→  1044→
   545→   545→  1045→  1045→  1045→    source.connect(gainNode)
   546→   546→  1046→  1046→  1046→    gainNode.connect(this.audioContext.destination)
   547→   547→  1047→  1047→  1047→    source.start()
   548→   548→  1048→  1048→  1048→
   549→   549→  1049→  1049→  1049→    source.onended = () => {
   550→   550→  1050→  1050→  1050→      gainNode.disconnect()
   551→   551→  1051→  1051→  1051→    }
   552→   552→  1052→  1052→  1052→  }
   553→   553→  1053→  1053→  1053→
   554→   554→  1054→  1054→  1054→  /**
   555→   555→  1055→  1055→  1055→   * Process an incoming chunk of audio from the microphone.
   556→   556→  1056→  1056→  1056→   * Accumulates into ring buffer, extracts chunks with context, sends to adapter.
   557→   557→  1057→  1057→  1057→   */
   558→   558→  1058→  1058→  1058→  private processInputChunk(inputData: Float32Array): void {
   559→   559→  1059→  1059→  1059→    // Append to ring buffer
   560→   560→  1060→  1060→  1060→    const newBuffer = new Float32Array(this.inputRingBuffer.length + inputData.length)
   561→   561→  1061→  1061→  1061→    newBuffer.set(this.inputRingBuffer)
   562→   562→  1062→  1062→  1062→    newBuffer.set(inputData, this.inputRingBuffer.length)
   563→   563→  1063→  1063→  1063→    this.inputRingBuffer = newBuffer
   564→   564→  1064→  1064→  1064→
   565→   565→  1065→  1065→  1065→    // Trim buffer if it exceeds max size
   566→   566→  1066→  1066→  1066→    if (this.inputRingBuffer.length > this.maxBufferSize) {
   567→   567→  1067→  1067→  1067→      const trimAmount = this.inputRingBuffer.length - this.maxBufferSize
   568→   568→  1068→  1068→  1068→      this.inputRingBuffer = this.inputRingBuffer.slice(trimAmount)
   569→   569→  1069→  1069→  1069→    }
   570→   570→  1070→  1070→  1070→
   571→   571→  1071→  1071→  1071→    // Check if we have enough audio for a chunk
   572→   572→  1072→  1072→  1072→    const chunkSamples = Math.ceil(this.config.chunkDuration * this.config.sourceSampleRate)
   573→   573→  1073→  1073→  1073→    const totalNeeded = chunkSamples
   574→   574→  1074→  1074→  1074→      + Math.ceil(this.config.leftContext * this.config.sourceSampleRate)
   575→   575→  1075→  1075→  1075→      + Math.ceil(this.config.rightContext * this.config.sourceSampleRate)
   576→   576→  1076→  1076→  1076→
   577→   577→  1077→  1077→  1077→    if (this.inputRingBuffer.length >= totalNeeded) {
   578→   578→  1078→  1078→  1078→      // Extract the chunk with context
   579→   579→  1079→  1079→  1079→      this.extractAndConvert(chunkSamples)
   580→   580→  1080→  1080→  1080→    }
   581→   581→  1081→  1081→  1081→  }
   582→   582→  1082→  1082→  1082→
   583→   583→  1083→  1083→  1083→  /**
   584→   584→  1084→  1084→  1084→   * Extract a chunk from the ring buffer with left/right context
   585→   585→  1085→  1085→  1085→   * and send it to the Seed-VC adapter for conversion.
   586→   586→  1086→  1086→  1086→   */
   587→   587→  1087→  1087→  1087→  private async extractAndConvert(chunkSamples: number): Promise<void> {
   588→   588→  1088→  1088→  1088→    const leftCtxSamples = Math.ceil(this.config.leftContext * this.config.sourceSampleRate)
   589→   589→  1089→  1089→  1089→    const rightCtxSamples = Math.ceil(this.config.rightContext * this.config.sourceSampleRate)
   590→   590→  1090→  1090→  1090→
   591→   591→  1091→  1091→  1091→    // The chunk is in the middle of the buffer
   592→   592→  1092→  1092→  1092→    const leftStart = this.inputRingBuffer.length - leftCtxSamples - chunkSamples - rightCtxSamples
   593→   593→  1093→  1093→  1093→    const chunkStart = leftStart + leftCtxSamples
   594→   594→  1094→  1094→  1094→    const chunkEnd = chunkStart + chunkSamples
   595→   595→  1095→  1095→  1095→
   596→   596→  1096→  1096→  1096→    if (chunkStart < 0 || chunkEnd > this.inputRingBuffer.length) return
   597→   597→  1097→  1097→  1097→
   598→   598→  1098→  1098→  1098→    // Extract just the chunk portion for conversion
   599→   599→  1099→  1099→  1099→    // (The adapter/referene audio handles the voice characteristics;
   600→   600→  1100→  1100→  1100→    // context is managed by Seed-VC internally for streaming mode)
   601→   601→  1101→  1101→  1101→    const chunk = this.inputRingBuffer.slice(chunkStart, chunkEnd)
   602→   602→  1102→  1102→  1102→
   603→   603→  1103→  1103→  1103→    // Noise gate check
   604→   604→  1104→  1104→  1104→    if (this.config.noiseGate) {
   605→   605→  1105→  1105→  1105→      const rms = this.calculateRMS(chunk)
   606→   606→  1106→  1106→  1106→      const db = 20 * Math.log10(rms + 1e-10)
   607→   607→  1107→  1107→  1107→      if (db < this.config.noiseGateThreshold) {
   608→   608→  1108→  1108→  1108→        // Below noise gate, skip conversion but advance buffer
   609→   609→  1109→  1109→  1109→        this.advanceBuffer(chunkSamples)
   610→   610→  1110→  1110→  1110→        return
   611→   611→  1111→  1111→  1111→      }
   612→   612→  1112→  1112→  1112→    }
   613→   613→  1113→  1113→  1113→
   614→   614→  1114→  1114→  1114→    try {
   615→   615→  1115→  1115→  1115→      // Send to adapter for conversion
   616→   616→  1116→  1116→  1116→      const converted = await this.adapter.convertChunk(chunk, this.config.sourceSampleRate)
   617→   617→  1117→  1117→  1117→
   618→   618→  1118→  1118→  1118→      // Notify via callback
   619→   619→  1119→  1119→  1119→      if (this.onAudioOutput) {
