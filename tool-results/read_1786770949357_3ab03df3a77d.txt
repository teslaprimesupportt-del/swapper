   136→def get_landmarks_5(face):
   137→    # face.landmark_5: np.ndarray shape (5,2)
   138→    # Если нет, попробуй face.kps или face.landmark
   139→    if hasattr(face, 'landmark_5') and face.landmark_5 is not None:
   140→        logger.debug("landmark_5: %s", face.landmark_5)
   141→        return face.landmark_5
   142→    elif hasattr(face, 'kps') and face.kps is not None:
   143→        logger.debug("kps: %s", face.kps)
   144→        return face.kps
   145→    elif hasattr(face, 'landmark') and face.landmark is not None:
   146→        # 68-точечная разметка, берём нужные индексы
   147→        # Иногда landmark shape (68,2) — тогда возьми нужные точки
   148→        # Пример: [36, 45, 30, 48, 54] — левый/правый глаз, нос, левый/правый рот

   150→            idxs = [36, 45, 30, 48, 54]
   151→            logger.debug("landmark (68 точек): %s", face.landmark[idxs])
   152→            return face.landmark[idxs]

   161→# Функция для вычисления аффинного преобразования
   162→def get_affine_transform(src_pts, dst_pts):
   163→    # src_pts, dst_pts: np.ndarray shape (5,2)
   164→    # OpenCV требует 3 точки, но можно использовать estimateAffinePartial2D для 5
   165→    M, _ = cv2.estimateAffinePartial2D(src_pts, dst_pts)
   166→    return M

   168→# Создаём градиентную маску овальной формы без обрезки
   169→# 2. Убедитесь, что эллипс **не выходит** за пределы 256×256  
   170→# Если эллипс "выпирает" за 256×256, `BORDER_CONSTANT` все равно создаст артефакты. Сократите размер эллипса, чтобы он полностью вписался в 256×256    
   171→def create_gradient_mask(crop_size=256):
   172→    # 1. Создаём пустую маску (все пиксели = 0)
   173→    mask = np.zeros((crop_size, crop_size), dtype=np.float32)
   174→    
   175→    # 2. Определяем центр и размеры эллипса (ИСПРАВЛЕНО: ещё меньше радиусов)
   176→    center = (crop_size // 2, crop_size // 2)
   177→    axes = (int(crop_size * 0.35), int(crop_size * 0.4))  # Уменьшили радиусы; Горизонтальный и вертикальный радиус
   178→    
   179→    # 3. Рисуем эллипс (заполняем белым цветом, значение=1.0)
   180→    cv2.ellipse(
   181→        mask,          # Массив для рисования
   182→        center,        # Центр эллипса
   183→        axes,          # Полуоси (ширина, высота)
   184→        angle=0,       # Угол поворота
   185→        startAngle=0,  # Начальный угол дуги
   186→        endAngle=360,  # Конечный угол дуги (360 = полный эллипс)
   187→        color=1.0,     # Значение для заполнения (белый = 1.0)
   188→        thickness=-1   # -1 = заполнить всю область эллипса    
   189→    )
   190→    
   191→    # 4. Применяем размытие для плавных краёв
   192→    blur_ksize = 15  # Нечётное число, чтобы ядро было симметричным
   193→    mask = cv2.GaussianBlur(mask, (blur_ksize, blur_ksize), 0)
   194→    
   195→    # 5. Ограничим значения в диапазоне [0, 1]
   196→    mask = np.clip(mask, 0, 1)
   197→    
   198→    return mask

   200→#### 1. Используйте `cv2.BORDER_TRANSPARENT` (OpenCV ≥ 4.5)  
   201→# Этот флаг позволяет **не заполнять** области за пределами маски никаким цветом (пиксели остаются `0` или "прозрачные").    
   202→def paste_back(target_img, swapped_face, M, crop_size=256):
   203→    # Улучшенная функция paste_back с идеальной овальной маской и исправлениями артефактов
   204→    
   205→    # target_img: Исходное изображение (BGR, numpy, uint8)
   206→    # swapped_face: Результат работы модели (256x256, BGR, uint8)
   207→    # M: Матрица аффинного преобразования (Target -> Crop), но здесь используется M_inv из run_hyperswap
   208→    # crop_size: Размер кропа (для HyperSwap это 256)
   209→    
   210→    # 1. Создание мягкой маски (Эрозия + Размытие)
   211→    mask = create_gradient_mask(crop_size)
   212→    
   213→    # Преобразуем в трехканальную маску
   214→    mask_3c = np.stack([mask] * 3, axis=2)
   215→    
   216→    # 2. Получаем размеры целевого изображения
   217→    h, w = target_img.shape[:2]
   218→    
   219→    # 3. Нормализация swapped_face к float32 [0,1] для warp
   220→    swapped_face_norm = swapped_face.astype(np.float32) / 255.0
   221→    mask_norm = mask_3c.astype(np.float32)  # Маска уже [0,1]
   222→    
   223→    # 4. Обратное преобразование (WARP_INVERSE_MAP) для лица И маски
   224→    # Используем BORDER_CONSTANT с borderValue=0.5 (серый, чтобы избежать синих/зеленых артефактов)
   225→    warped_face = cv2.warpAffine(
   226→        swapped_face_norm,
   227→        M,  # Это M_inv из run_hyperswap
   228→        (w, h),
   229→        flags=cv2.INTER_LANCZOS4 | cv2.WARP_INVERSE_MAP,
   230→        borderMode=cv2.BORDER_CONSTANT,
   231→        borderValue=0.5  # Серый фон вместо черного/белого
   232→    )
   233→    
   234→    warped_mask = cv2.warpAffine(
   235→        mask_norm,
   236→        M,  # Это M_inv из run_hyperswap
   237→        (w, h),
   238→        flags=cv2.INTER_CUBIC | cv2.WARP_INVERSE_MAP,
   239→        borderMode=cv2.BORDER_CONSTANT,
   240→        borderValue=0.0  # Маска: 0 за пределами
   241→    )
   242→    
   243→    # 5. Обработка после warp: Clip, NaN fix
   244→    warped_face = np.clip(warped_face, 0, 1)  # Убираем отрицательные
   245→    warped_face = np.nan_to_num(warped_face, nan=0.5)  # NaN -> серый
   246→    
   247→    warped_mask = np.clip(warped_mask, 0, 1)
   248→    warped_mask = np.nan_to_num(warped_mask, nan=0.0)
   249→    
   250→    # 6. Дополнительное размытие для устранения артефактов (опционально, но помогает)
   251→    warped_mask = cv2.GaussianBlur(warped_mask, (3, 3), 0)
   252→    
   253→    # Отладочные логи (добавьте после warp)
   254→    logger.debug("Warped face shape: %s | Min: %s | Max: %s | NaN count: %s", 
   255→                 warped_face.shape, warped_face.min(), warped_face.max(), np.isnan(warped_face).sum())
   256→    logger.debug("Warped mask shape: %s | Min: %s | Max: %s | NaN count: %s", 
   257→                 warped_mask.shape, warped_mask.min(), warped_mask.max(), np.isnan(warped_mask).sum())
   258→    
   259→    # 7. Плавное наложение в float32
   260→    target_float = target_img.astype(np.float32) / 255.0
   261→    result_float = target_float * (1.0 - warped_mask) + warped_face * warped_mask
   262→    
   263→    # 8. Обратная нормализация к uint8
   264→    result = (result_float * 255).clip(0, 255).astype(np.uint8)
   265→    
   266→    logger.debug("Final result: shape %s | Min: %s | Max: %s", result.shape, result.min(), result.max())
   267→    
   268→    return result

   282→# Итоговая функция run_hyperswap с аффинным преобразованием
   283→def run_hyperswap(session, source_face, target_face, target_img):
   284→    # 1. Подготовка эмбеддинга
   285→    source_embedding = source_face.normed_embedding.reshape(1, -1).astype(np.float32)
   286→
   287→    # 2. Получаем 5 точек target
   288→    target_landmarks_5 = get_landmarks_5(target_face)
   289→    visualize_points(target_img, target_landmarks_5, (0, 255, 0))  # Зеленые точки
   290→    
   291→    if target_landmarks_5 is None:
   292→        logger.error("Не удалось получить 5 точек для целевого лица")
   293→        # Важно: Если ошибка, возвращаем None и исходную матрицу (или обрабатываем ошибку иначе)
   294→        return None, None
   295→
   296→    # 3. Определение эталонных точек для выравнивания 256x256 (FFHQ Alignment)
   297→    std_landmarks_256 = np.array([
   298→        [ 84.87, 105.94],  # Левый глаз
   299→        [171.13, 105.94],  # Правый глаз
   300→        [128.00, 146.66],  # Кончик носа
   301→        [ 96.95, 188.64],  # Левый уголок рта
   302→        [159.05, 188.64]   # Правый уголок рта
   303→    ], dtype=np.float32)
   304→
   305→    # Вычисляем аффинную матрицу
   306→    M = get_affine_transform(target_landmarks_5.astype(np.float32), std_landmarks_256)
   307→    logger.debug("Affine Matrix M (used for cropping):\n%s", M)
   308→
   309→#### Что проверить:
   310→# Матрица `M` не должна содержать `NaN` или бесконечности.
   311→# Если матрица нулевая или искаженная — проблема в точках `target_landmarks_5`.
   312→    
   313→    # Применяем аффинное преобразование с новой матрицей M
   314→    crop = cv2.warpAffine(target_img, M, (256, 256), flags=cv2.INTER_CUBIC, borderMode=cv2.BORDER_REFLECT)
   315→
   316→    # Визуализация crop перед инференсом
   317→    #### Что проверить:
   318→    # Окно `"Crop Before Inference"` должно показывать лицо, вырезанное по аффинному преобразованию.
   319→    # Если изображение черное — проблема в `M` или `target_landmarks_5`.
   320→    logger.debug("Crop shape: %s | Min: %s | Max: %s", crop.shape, crop.min(), crop.max())
   321→  # cv2.imshow("Crop Before Inference", crop)
   322→  # cv2.waitKey(1)  # Отображает изображение
   323→
   324→    # 4. Преобразуем crop для модели
   325→    # crop_input = crop[:, :, ::-1] / 255.0
   326→    # crop_input = (crop_input - 0.5) / 0.5
   327→    # crop_input = crop_input.transpose(2, 0, 1)
   328→    # crop_input = np.expand_dims(crop_input, axis=0).astype(np.float32)
   329→    crop_input = crop[:, :, ::-1].astype(np.float32) / 255.0  # RGB -> [0,1]
   330→    crop_input = (crop_input - 0.5) / 0.5  # Нормализация
   331→    crop_input = crop_input.transpose(2, 0, 1)[np.newaxis, ...].astype(np.float32)
   332→
   333→    # 5. Инференс
   334→    try:
   335→        output = session.run(None, {'source': source_embedding, 'target': crop_input})[0][0]
   336→        logger.debug("Model output shape: %s | Min: %s | Max: %s", output.shape, output.min(), output.max())
   337→    except Exception as e:
   338→        logger.error("Ошибка выполнения модели: %s", e)
   339→        return target_img
   340→
   341→    # --- CPU FLOAT NORMALIZATION FIX ---
   342→    # предотвращает появление "синей кожи" и "шума" при работе на CPU
   343→    if isinstance(output, np.ndarray):
   344→        # устранение NaN и бесконечностей
   345→        output = np.nan_to_num(output, nan=0.0, posinf=255.0, neginf=0.0)
   346→        
   347→        # если диапазон похож на [-1,1] → нормализуем в [0,255]
   348→        if output.min() < 0.0 or output.max() <= 1.5:
   349→            output = ((output + 1.0) / 2.0 * 255.0)
   350→        
   351→        # жёсткое ограничение диапазона и тип для OpenCV
   352→        output = np.clip(output, 0, 255).astype(np.uint8).copy()
   353→        
   354→        # защита от повторного использования буфера (inplace CPU bug)
   355→        try:
   356→            output.setflags(write=True)
   357→        except Exception:
   358→            pass
   359→
   360→    # 6. Обратная нормализация (теперь output уже uint8, просто transpose и BGR)
   361→    # (ваш код без изменений, но без старой денормализации)
   362→    output = output.transpose(1, 2, 0)  # CHW -> HWC
   363→    output = output[:, :, ::-1]  # RGB -> BGR (Убедитесь, что это BGR, если вход был BGR)
   364→    logger.debug("Output after denormalization: Min: %s | Max: %s", output.min(), output.max())
   365→    
   366→    # Визуализация после денормализации
   367→    #### Что проверить:
   368→    # `output` должен быть в диапазоне `[0..255]` и содержать лицо.
   369→    # Если `output` черный — проблема в нормализации/денормализации или в самой модели.
   370→    logger.debug("Output after denormalization: Min: %s | Max: %s", output.min(), output.max())
   371→  # cv2.imshow("Output After Denormalization", output)
   372→  # cv2.waitKey(1)
   373→       
   374→    return output, M # Возвращаем лицо (256x256) и матрицу M

   376→def sort_by_order(face, order: str):
   377→    if order == "left-right":
   378→        return sorted(face, key=lambda x: x.bbox[0])
   379→    if order == "right-left":
   380→        return sorted(face, key=lambda x: x.bbox[0], reverse = True)
   381→    if order == "top-bottom":
   382→        return sorted(face, key=lambda x: x.bbox[1])
   383→    if order == "bottom-top":
   384→        return sorted(face, key=lambda x: x.bbox[1], reverse = True)
   385→    if order == "small-large":
   386→        return sorted(face, key=lambda x: (x.bbox[2] - x.bbox[0]) * (x.bbox[3] - x.bbox[1]))
   387→    # if order == "large-small":
   388→    #     return sorted(face, key=lambda x: (x.bbox[2] - x.bbox[0]) * (x.bbox[3] - x.bbox[1]), reverse = True)
   389→    # by default "large-small":
   390→    return sorted(face, key=lambda x: (x.bbox[2] - x.bbox[0]) * (x.bbox[3] - x.bbox[1]), reverse = True)

   392→def get_face_gender(
   393→        face,
   394→        face_index,
   395→        gender_condition,
   396→        operated: str,
   397→        order: str,
   398→):
   399→    gender = [
   400→        x.sex
   401→        for x in face
   402→    ]
   403→    gender.reverse()
