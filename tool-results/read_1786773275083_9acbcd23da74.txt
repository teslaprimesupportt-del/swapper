    35→function formatDuration(seconds: number): string {
    36→  const h = Math.floor(seconds / 3600)
    37→  const m = Math.floor((seconds % 3600) / 60)
    38→  const s = seconds % 60
    39→  if (h > 0) return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    40→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    41→}

   485→function SettingsDialog() {
   486→  const { providers, setProviders, activeProvider, activeFaceProvider, setActiveProvider, setActiveFaceProvider, updateProviderStatus } = useStudioStore()
   487→  const [name, setName] = useState('')
   488→  const [type, setType] = useState<ProviderType>('seed-vc')
   489→  const [endpoint, setEndpoint] = useState('')
   490→  const [apiKey, setApiKey] = useState('')
   491→
   492→  const addProvider = () => {
   493→    if (!name || !endpoint) return
   494→    const newProvider = {
   495→      id: crypto.randomUUID(),
   496→      name, type, endpoint, apiKey,
   497→      status: 'disconnected' as const,
   498→    }
   499→    setProviders([...providers, newProvider])
   500→    setName('')
   501→    setEndpoint('')
   502→    setApiKey('')
   503→  }
   504→
   505→  const testProvider = async (id: string) => {
   506→    const provider = providers.find(p => p.id === id)
   507→    if (!provider) return
   508→    updateProviderStatus(id, 'connecting')
   509→    try {
   510→      const controller = new AbortController()
   511→      const timeout = setTimeout(() => controller.abort(), 5000)
   512→      await fetch(provider.endpoint, {
   513→        method: 'GET',
   514→        signal: controller.signal,
   515→        headers: provider.apiKey ? { Authorization: `Bearer ${provider.apiKey}` } : {},
   516→      })
   517→      clearTimeout(timeout)
   518→      updateProviderStatus(id, 'connected')
   519→    } catch {
   520→      updateProviderStatus(id, 'error')
   521→    }
   522→  }
   523→
   524→  return (
   525→    <Dialog>
   526→      <DialogTrigger asChild>
   527→        <Button variant="ghost" size="icon" className="w-8 h-8">
   528→          <Settings className="w-4 h-4" />
