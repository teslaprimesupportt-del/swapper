    35→    35→function formatDuration(seconds: number): string {
    36→    36→  const h = Math.floor(seconds / 3600)
    37→    37→  const m = Math.floor((seconds % 3600) / 60)
    38→    38→  const s = seconds % 60
    39→    39→  if (h > 0) return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    40→    40→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    41→    41→}

   485→   485→function SettingsDialog() {

   487→   487→  const [name, setName] = useState('')

   489→   489→  const [endpoint, setEndpoint] = useState('')
   490→   490→  const [apiKey, setApiKey] = useState('')
   491→   491→
   492→   492→  const addProvider = () => {
   493→   493→    if (!name || !endpoint) return
   494→   494→    const newProvider = {
   495→   495→      id: crypto.randomUUID(),
   496→   496→      name, type, endpoint, apiKey,
   497→   497→      status: 'disconnected' as const,
   498→   498→    }
   499→   499→    setProviders([...providers, newProvider])
   500→   500→    setName('')
   501→   501→    setEndpoint('')
   502→   502→    setApiKey('')
   503→   503→  }
   504→   504→
   505→   505→  const testProvider = async (id: string) => {
   506→   506→    const provider = providers.find(p => p.id === id)
   507→   507→    if (!provider) return
   508→   508→    updateProviderStatus(id, 'connecting')
   509→   509→    try {
   510→   510→      const controller = new AbortController()
   511→   511→      const timeout = setTimeout(() => controller.abort(), 5000)
   512→   512→      await fetch(provider.endpoint, {
   513→   513→        method: 'GET',
   514→   514→        signal: controller.signal,
   515→   515→        headers: provider.apiKey ? { Authorization: `Bearer ${provider.apiKey}` } : {},
   516→   516→      })
   517→   517→      clearTimeout(timeout)
   518→   518→      updateProviderStatus(id, 'connected')
   519→   519→    } catch {
   520→   520→      updateProviderStatus(id, 'error')
   521→   521→    }
   522→   522→  }
   523→   523→
