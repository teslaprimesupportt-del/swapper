    35→function formatDuration(seconds: number): string {
    36→  const h = Math.floor(seconds / 3600)
    37→  const m = Math.floor((seconds % 3600) / 60)
    38→  const s = seconds % 60
    39→  if (h > 0) return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    40→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    41→}

   516→      <div className="flex border-b border-studio-border/50 shrink-0">
   517→        {tabs.map((tab) => (

   520→            onClick={() => setActiveTab(tab.key)}
   521→            className={cn(
   522→              'flex-1 py-2.5 text-xs font-medium uppercase tracking-wider transition-colors',

   545→function SettingsDialog() {
   546→  const { providers, setProviders, activeProvider, activeFaceProvider, setActiveProvider, setActiveFaceProvider, updateProviderStatus } = useStudioStore()
   547→  const [name, setName] = useState('')
   548→  const [type, setType] = useState<ProviderType>('seed-vc')
   549→  const [endpoint, setEndpoint] = useState('')
   550→  const [apiKey, setApiKey] = useState('')
   551→
   552→  const addProvider = () => {
   553→    if (!name || !endpoint) return
   554→    const newProvider = {
   555→      id: crypto.randomUUID(),
   556→      name, type, endpoint, apiKey,
   557→      status: 'disconnected' as const,
   558→    }
   559→    setProviders([...providers, newProvider])
   560→    setName('')
   561→    setEndpoint('')
   562→    setApiKey('')
   563→  }
   564→
   565→  const testProvider = async (id: string) => {
   566→    const provider = providers.find(p => p.id === id)
   567→    if (!provider) return
   568→    updateProviderStatus(id, 'connecting')
   569→    try {
   570→      const controller = new AbortController()
   571→      const timeout = setTimeout(() => controller.abort(), 5000)
   572→      await fetch(provider.endpoint, {
   573→        method: 'GET',
   574→        signal: controller.signal,
   575→        headers: provider.apiKey ? { Authorization: `Bearer ${provider.apiKey}` } : {},
   576→      })
   577→      clearTimeout(timeout)
   578→      updateProviderStatus(id, 'connected')
   579→    } catch {
   580→      updateProviderStatus(id, 'error')
   581→    }
   582→  }
   583→
   584→  return (
   585→    <Dialog>
   586→      <DialogTrigger asChild>
   587→        <Button variant="ghost" size="icon" className="w-8 h-8">
   588→          <Settings className="w-4 h-4" />
   589→        </Button>
   590→      </DialogTrigger>
   591→      <DialogContent className="sm:max-w-md bg-card border-studio-border/50">
   592→        <DialogHeader>
   593→          <DialogTitle className="flex items-center gap-2">
   594→            <Settings className="w-4 h-4" />
   595→            AI Provider Settings (BYOK)
   596→          </DialogTitle>
   597→        </DialogHeader>
   598→
   599→        <div className="space-y-4 mt-4">
   600→          {/* Add Provider Form */}
   601→          <div className="space-y-3 p-3 rounded-lg border border-studio-border/30 bg-secondary/30">
   602→            <p className="text-xs font-medium uppercase tracking-wider text-studio-muted-foreground/70">
   603→              Add Provider
   604→            </p>
   605→            <div className="grid gap-2">
   606→              <Input
   607→                placeholder="Provider name (e.g. My Seed-VC Server)"
   608→                value={name}
   609→                onChange={(e) => setName(e.target.value)}
   610→                className="h-9 text-sm"
   611→              />
   612→              <Select value={type} onValueChange={(v) => setType(v as ProviderType)}>
   613→                <SelectTrigger className="h-9 text-sm">
   614→                  <SelectValue />
   615→                </SelectTrigger>
   616→                <SelectContent>
   617→                  <SelectItem value="seed-vc">Seed-VC (Zero-Shot VC) - Recommended</SelectItem>
   618→                  <SelectItem value="rvc">RVC (Voice Conversion)</SelectItem>
   619→                  <SelectItem value="openvoice">OpenVoice (Voice Clone)</SelectItem>
   620→                  <SelectItem value="liveportrait">LivePortrait (Face)</SelectItem>
   621→                  <SelectItem value="faceswap">Face Swap (ComfyUI ReActor)</SelectItem>
   622→                  <SelectItem value="wav2lip">Wav2Lip (Lip Sync)</SelectItem>
   623→                  <SelectItem value="custom">Custom</SelectItem>
   624→                </SelectContent>
   625→              </Select>
   626→              <Input
   627→                placeholder="e.g. https://xxxxx.gradio.live or http://127.0.0.1:8188"
   628→                value={endpoint}
   629→                onChange={(e) => setEndpoint(e.target.value)}
   630→                className="h-9 text-sm"
   631→              />

   633→                placeholder="API Key (optional - most instances don't need one)"

   635→                value={apiKey}
   636→                onChange={(e) => setApiKey(e.target.value)}
   637→                className="h-9 text-sm"
