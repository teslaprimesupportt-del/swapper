    36→    36→function formatDuration(seconds: number): string {
    37→    37→  const h = Math.floor(seconds / 3600)
    38→    38→  const m = Math.floor((seconds % 3600) / 60)
    39→    39→  const s = seconds % 60
    40→    40→  if (h > 0) return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    41→    41→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    42→    42→}

   549→   549→function SettingsDialog() {
   550→   550→  const { providers, setProviders, activeProvider, activeFaceProvider, setActiveProvider, setActiveFaceProvider, updateProviderStatus, isSettingsOpen, setSettingsOpen } = useStudioStore()
   551→   551→  const [name, setName] = useState('')
   552→   552→  const [type, setType] = useState<ProviderType>('seed-vc')
   553→   553→  const [endpoint, setEndpoint] = useState('')
   554→   554→  const [apiKey, setApiKey] = useState('')
   555→   555→  const [editingId, setEditingId] = useState<string | null>(null)
   556→   556→
   557→   557→  const resetForm = () => {
   558→   558→    setName('')
   559→   559→    setType('seed-vc')
   560→   560→    setEndpoint('')
   561→   561→    setApiKey('')
   562→   562→    setEditingId(null)
   563→   563→  }
   564→   564→
   565→   565→  const addOrUpdateProvider = () => {
   566→   566→    if (!name || !endpoint) return
   567→   567→    if (editingId) {
   568→   568→      // Update existing
   569→   569→      setProviders(providers.map(p =>
   570→   570→        p.id === editingId ? { ...p, name, type, endpoint, apiKey } : p
   571→   571→      ))
   572→   572→    } else {
   573→   573→      // Add new
   574→   574→      const newProvider = {
   575→   575→        id: crypto.randomUUID(),
   576→   576→        name, type, endpoint, apiKey,
   577→   577→        status: 'disconnected' as const,
   578→   578→      }
   579→   579→      setProviders([...providers, newProvider])
   580→   580→    }
   581→   581→    resetForm()
   582→   582→  }
   583→   583→
   584→   584→  const editProvider = (id: string) => {
   585→   585→    const p = providers.find(x => x.id === id)
   586→   586→    if (!p) return
   587→   587→    setEditingId(id)
   588→   588→    setName(p.name)
   589→   589→    setType(p.type)
   590→   590→    setEndpoint(p.endpoint)
   591→   591→    setApiKey(p.apiKey)
   592→   592→  }
   593→   593→
   594→   594→  const deleteProvider = (id: string) => {
   595→   595→    setProviders(providers.filter(p => p.id !== id))
   596→   596→    if (activeProvider?.id === id) setActiveProvider(null)
   597→   597→    if (activeFaceProvider?.id === id) setActiveFaceProvider(null)
   598→   598→  }
   599→   599→
   600→   600→  const testProvider = async (id: string) => {
   601→   601→    const provider = providers.find(p => p.id === id)
   602→   602→    if (!provider) return
   603→   603→    updateProviderStatus(id, 'connecting')
   604→   604→    try {
   605→   605→      const controller = new AbortController()
   606→   606→      const timeout = setTimeout(() => controller.abort(), 5000)
   607→   607→      await fetch(provider.endpoint, {
   608→   608→        method: 'GET',
   609→   609→        signal: controller.signal,
   610→   610→        headers: provider.apiKey ? { Authorization: `Bearer ${provider.apiKey}` } : {},
   611→   611→      })
   612→   612→      clearTimeout(timeout)
   613→   613→      updateProviderStatus(id, 'connected')
   614→   614→    } catch {
   615→   615→      updateProviderStatus(id, 'error')
   616→   616→    }
   617→   617→  }
   618→   618→
   619→   619→  const typeHint = type === 'faceswap'
   620→   620→    ? 'Point to a ComfyUI instance with ReActor installed (e.g. https://xxx.trycloudflare.com or http://127.0.0.1:8188)'
   621→   621→    : type === 'seed-vc'
   622→   622→      ? 'Point to a Seed-VC Gradio instance (e.g. https://xxxxx.gradio.live)'
   623→   623→      : 'Enter the API endpoint URL for this provider'
   624→   624→
   625→   625→  return (
   626→   626→    <Dialog open={isSettingsOpen} onOpenChange={(open) => setSettingsOpen(open)}>
   627→   627→      <DialogTrigger asChild>
   628→   628→        <Button variant="ghost" size="icon" className="w-8 h-8">
   629→   629→          <Settings className="w-4 h-4" />
   630→   630→        </Button>
   631→   631→      </DialogTrigger>
   632→   632→      <DialogContent className="sm:max-w-md bg-card border-studio-border/50">
   633→   633→        <DialogHeader>
   634→   634→          <DialogTitle className="flex items-center gap-2">
   635→   635→            <Settings className="w-4 h-4" />
   636→   636→            AI Provider Settings (BYOK)
   637→   637→          </DialogTitle>
   638→   638→        </DialogHeader>
   639→   639→
   640→   640→        <div className="space-y-4 mt-4">
   641→   641→          {/* Add / Edit Provider Form */}
   642→   642→          <div className="space-y-3 p-3 rounded-lg border border-studio-border/30 bg-secondary/30">
   643→   643→            <p className="text-xs font-medium uppercase tracking-wider text-studio-muted-foreground/70">
   644→   644→              {editingId ? 'Edit Provider' : 'Add Provider'}
   645→   645→            </p>
   646→   646→            <div className="grid gap-2">
   647→   647→              <Input
   648→   648→                placeholder="Provider name (e.g. My Seed-VC Server)"
   649→   649→                value={name}
   650→   650→                onChange={(e) => setName(e.target.value)}
   651→   651→                className="h-9 text-sm"
   652→   652→              />
   653→   653→              <Select value={type} onValueChange={(v) => setType(v as ProviderType)}>
   654→   654→                <SelectTrigger className="h-9 text-sm">
   655→   655→                  <SelectValue />
   656→   656→                </SelectTrigger>
   657→   657→                <SelectContent>
   658→   658→                  <SelectItem value="seed-vc">Seed-VC (Zero-Shot VC) - Voice</SelectItem>
   659→   659→                  <SelectItem value="rvc">RVC (Voice Conversion)</SelectItem>
   660→   660→                  <SelectItem value="openvoice">OpenVoice (Voice Clone)</SelectItem>
   661→   661→                  <SelectItem value="liveportrait">LivePortrait (Face)</SelectItem>
   662→   662→                  <SelectItem value="faceswap">Face Swap (ComfyUI ReActor) - Face</SelectItem>
   663→   663→                  <SelectItem value="wav2lip">Wav2Lip (Lip Sync)</SelectItem>
   664→   664→                  <SelectItem value="custom">Custom</SelectItem>
   665→   665→                </SelectContent>
   666→   666→              </Select>
   667→   667→              <Input
   668→   668→                placeholder={typeHint}
   669→   669→                value={endpoint}
   670→   670→                onChange={(e) => setEndpoint(e.target.value)}
   671→   671→                className="h-9 text-sm"
   672→   672→              />
   673→   673→              <Input
   674→   674→                placeholder="API Key (optional - most instances don't need one)"
   675→   675→                type="password"
   676→   676→                value={apiKey}
   677→   677→                onChange={(e) => setApiKey(e.target.value)}
   678→   678→                className="h-9 text-sm"
   679→   679→              />
   680→   680→              {type === 'faceswap' && (
   681→   681→                <div className="space-y-1">

   766→   766→                        variant="ghost"
   767→   767→                        size="sm"

   769→   769→                          'h-7 text-xs',

   778→   778→                        variant="ghost"
