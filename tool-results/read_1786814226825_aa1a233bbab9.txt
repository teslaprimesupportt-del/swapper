     1→    36→    36→function formatDuration(seconds: number): string {
     2→    37→    37→  const h = Math.floor(seconds / 3600)
     3→    38→    38→  const m = Math.floor((seconds % 3600) / 60)
     4→    39→    39→  const s = seconds % 60
     5→    40→    40→  if (h > 0) return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
     6→    41→    41→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
     7→    42→    42→}
