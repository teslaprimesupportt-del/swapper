   549→function SettingsDialog() {

   551→  const [name, setName] = useState('')

   555→  const [editingId, setEditingId] = useState<string | null>(null)
   556→
   557→  const resetForm = () => {
   558→    setName('')
   559→    setType('seed-vc')
   560→    setEndpoint('')
   561→    setApiKey('')
   562→    setEditingId(null)
   563→  }
   564→
   565→  const addOrUpdateProvider = () => {
   566→    if (!name || !endpoint) return
   567→    if (editingId) {
   568→      // Update existing
   569→      setProviders(providers.map(p =>
   570→        p.id === editingId ? { ...p, name, type, endpoint, apiKey } : p
   571→      ))
   572→    } else {
   573→      // Add new
   574→      const newProvider = {
   575→        id: crypto.randomUUID(),
   576→        name, type, endpoint, apiKey,
   577→        status: 'disconnected' as const,
   578→      }
   579→      setProviders([...providers, newProvider])
   580→    }
   581→    resetForm()
   582→  }
   583→
   584→  const editProvider = (id: string) => {
   585→    const p = providers.find(x => x.id === id)
   586→    if (!p) return
   587→    setEditingId(id)
   588→    setName(p.name)
   589→    setType(p.type)
   590→    setEndpoint(p.endpoint)
   591→    setApiKey(p.apiKey)
   592→  }
   593→
   594→  const deleteProvider = (id: string) => {
   595→    setProviders(providers.filter(p => p.id !== id))

   597→    if (activeFaceProvider?.id === id) setActiveFaceProvider(null)

   599→
   600→  const testProvider = async (id: string) => {
   601→    const provider = providers.find(p => p.id === id)
   602→    if (!provider) return
   603→    updateProviderStatus(id, 'connecting')
   604→    try {
   605→      const controller = new AbortController()
   606→      const timeout = setTimeout(() => controller.abort(), 5000)
   607→      await fetch(provider.endpoint, {
   608→        method: 'GET',
   609→        signal: controller.signal,
   610→        headers: provider.apiKey ? { Authorization: `Bearer ${provider.apiKey}` } : {},
   611→      })
   612→      clearTimeout(timeout)
   613→      updateProviderStatus(id, 'connected')
   614→    } catch {
   615→      updateProviderStatus(id, 'error')
   616→    }
   617→  }
   618→
   619→  const typeHint = type === 'faceswap'
   620→    ? 'Point to a ComfyUI instance with ReActor installed (e.g. https://xxx.trycloudflare.com or http://127.0.0.1:8188)'
   621→    : type === 'seed-vc'
   622→      ? 'Point to a Seed-VC Gradio instance (e.g. https://xxxxx.gradio.live)'
   623→      : 'Enter the API endpoint URL for this provider'
   624→
   625→  return (
   626→    <Dialog open={isSettingsOpen} onOpenChange={(open) => setSettingsOpen(open)}>
   627→      <DialogTrigger asChild>
   628→        <Button variant="ghost" size="icon" className="w-8 h-8">
   629→          <Settings className="w-4 h-4" />
   630→        </Button>
   631→      </DialogTrigger>
   632→      <DialogContent className="sm:max-w-md bg-card border-studio-border/50">
   633→        <DialogHeader>
   634→          <DialogTitle className="flex items-center gap-2">
   635→            <Settings className="w-4 h-4" />
   636→            AI Provider Settings (BYOK)
   637→          </DialogTitle>
   638→        </DialogHeader>
   639→
