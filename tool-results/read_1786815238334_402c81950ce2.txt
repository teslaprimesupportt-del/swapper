    36→    36→function formatDuration(seconds: number): string {
    37→    37→  const h = Math.floor(seconds / 3600)
    38→    38→  const m = Math.floor((seconds % 3600) / 60)
    39→    39→  const s = seconds % 60
    40→    40→  if (h > 0) return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    41→    41→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    42→    42→}
